```ts
import { gzipSync, gunzipSync } from "node:zlib";
import { readFileSync, writeFileSync } from "node:fs";
import { DOMParser, XMLSerializer } from "@xmldom/xmldom";
import type { Document, Element } from "@xmldom/xmldom";

export type { Document, Element };

const GZIP_MAGIC = 0x1f;

export function loadAlsBuffer(buf: Buffer): Document {
  const xml = buf[0] === GZIP_MAGIC ? gunzipSync(buf).toString("utf-8") : buf.toString("utf-8");
  return new DOMParser().parseFromString(xml, "text/xml");
}

export function loadAls(path: string): Document {
  return loadAlsBuffer(readFileSync(path));
}

export function serialize(doc: Document): string {
  return '<?xml version="1.0" encoding="UTF-8"?>\n' +
    new XMLSerializer().serializeToString(doc.documentElement!);
}

export function saveAls(doc: Document, path: string): void {
  writeFileSync(path, gzipSync(Buffer.from(serialize(doc), "utf-8")));
}

export function allEl(root: Document | Element, tag: string): Element[] {
  return Array.from(root.getElementsByTagName(tag)) as Element[];
}

export function pathEl(el: Element, tags: string[]): Element | null {
  let cur: Element | null = el;
  for (const tag of tags) {
    if (!cur) return null;
    let next: Element | null = null;
    for (let n = cur.firstChild; n; n = n.nextSibling) {
      if (n.nodeType === 1 && (n as Element).tagName === tag) { next = n as Element; break; }
    }
    cur = next;
  }
  return cur;
}

export function valueOf(el: Element | null, fallback = ""): string {
  return el?.getAttribute("Value") ?? fallback;
}

export function pathVal(el: Element, tags: string[]): string {
  return valueOf(pathEl(el, tags));
}
```

```ts
import { DOMParser } from "@xmldom/xmldom";
import type { Element } from "@xmldom/xmldom";

export interface PlistDict { [k: string]: PlistValue }
export type PlistValue = string | number | boolean | Buffer | PlistValue[] | PlistDict;

export function parseXmlPlist(buf: Buffer): PlistValue {
  const head = buf.subarray(0, 64).toString("latin1");
  if (!head.includes("<?xml") && !head.includes("<plist")) throw new Error("not an XML plist");
  const doc = new DOMParser().parseFromString(buf.toString("utf-8"), "text/xml");
  const root = firstElChild(doc.getElementsByTagName("plist")[0] as Element);
  if (!root) throw new Error("empty plist");
  return convert(root);
}

function firstElChild(el: Element | undefined): Element | null {
  for (let n = el?.firstChild; n; n = n.nextSibling) if (n.nodeType === 1) return n as Element;
  return null;
}

function elChildren(el: Element): Element[] {
  const out: Element[] = [];
  for (let n = el.firstChild; n; n = n.nextSibling) if (n.nodeType === 1) out.push(n as Element);
  return out;
}

function convert(el: Element): PlistValue {
  switch (el.tagName) {
    case "string": return el.textContent ?? "";
    case "integer": return parseInt(el.textContent ?? "0", 10);
    case "real": return parseFloat(el.textContent ?? "0");
    case "true": return true;
    case "false": return false;
    case "data": return Buffer.from((el.textContent ?? "").replace(/\s/g, ""), "base64");
    case "array": return elChildren(el).map(convert);
    case "dict": {
      const out: PlistDict = {};
      const kids = elChildren(el);
      for (let i = 0; i + 1 < kids.length; i += 2) {
        if (kids[i].tagName === "key") out[kids[i].textContent ?? ""] = convert(kids[i + 1]);
      }
      return out;
    }
    default: return el.textContent ?? "";
  }
}
```

`src/devices.ts`:
```ts
import { allEl, pathEl, pathVal, valueOf } from "./als.ts";
import type { Document, Element } from "./als.ts";
import { parseXmlPlist } from "./plist.ts";
import type { PlistDict } from "./plist.ts";

export type DeviceKind = "au" | "vst3" | "vst2" | "native";
export interface DeviceRef { kind: DeviceKind; name: string; trackName: string; el: Element }

const TRACK_TAGS = ["MidiTrack", "AudioTrack", "ReturnTrack", "GroupTrack", "MainTrack"];

export function scanDevices(doc: Document): DeviceRef[] {
  const out: DeviceRef[] = [];
  for (const tag of TRACK_TAGS) {
    for (const track of allEl(doc, tag)) {
      const trackName = pathVal(track, ["Name", "EffectiveName"]) || "(unnamed)";
      for (const dev of allEl(track, "AuPluginDevice")) {
        out.push({ kind: "au", name: pathVal(dev, ["PluginDesc", "AuPluginInfo", "Name"]), trackName, el: dev });
      }
      for (const dev of allEl(track, "PluginDevice")) {
        const v3 = pathEl(dev, ["PluginDesc", "Vst3PluginInfo"]);
        const v2 = pathEl(dev, ["PluginDesc", "VstPluginInfo"]);
        if (v3) out.push({ kind: "vst3", name: pathVal(dev, ["PluginDesc", "Vst3PluginInfo", "Name"]), trackName, el: dev });
        else if (v2) out.push({ kind: "vst2", name: valueOf(pathEl(v2, ["PlugName"])), trackName, el: dev });
      }
    }
  }
  return out;
}

export function auState(dev: DeviceRef): PlistDict | null {
  for (const buf of allEl(dev.el, "Buffer")) {
    const hex = (buf.textContent ?? "").replace(/\s/g, "");
    if (hex.length < 32) continue;
    try {
      const v = parseXmlPlist(Buffer.from(hex, "hex"));
      if (v && typeof v === "object" && !Buffer.isBuffer(v) && !Array.isArray(v)) return v as PlistDict;
    } catch { return null; } // binary plist etc. — report upstream, don't guess
  }
  return null;
}

export function auCodes(state: PlistDict) {
  return {
    type: Number(state.type ?? 0),
    subtype: Number(state.subtype ?? 0),
    manufacturer: Number(state.manufacturer ?? 0),
  };
}

export function fourcc(n: number): string {
  const u = n >>> 0;
  return String.fromCharCode((u >>> 24) & 0xff, (u >>> 16) & 0xff, (u >>> 8) & 0xff, u & 0xff);
}

export function getProcessorState(deviceEl: Element): Buffer | null {
  const ps = allEl(deviceEl, "ProcessorState")[0];
  if (!ps) return null;
  return Buffer.from((ps.textContent ?? "").replace(/\s/g, ""), "hex");
}

export function setProcessorState(deviceEl: Element, state: Buffer): void {
  const ps = allEl(deviceEl, "ProcessorState")[0];
  if (!ps) throw new Error("device has no ProcessorState element");
  ps.textContent = state.toString("hex").toUpperCase();
}

export function liveDeviceType(auTypeFourcc: string): number {
  return auTypeFourcc === "aumu" ? 1 : 2; // instrument vs audio effect
}
```

`src/pointee.ts`:
```ts
import { allEl } from "./als.ts";
import type { Document, Element } from "./als.ts";

const POINTEE_TAGS = ["AutomationTarget", "ModulationTarget", "Pointee"];

// Live invariant (learned the hard way): every pointee-space Id must be unique
// AND below LiveSet/NextPointeeId, or Live reports "document is corrupt".
export function renumberPointees(deviceEl: Element, doc: Document): number {
  const counter = allEl(doc, "NextPointeeId")[0];
  if (!counter) throw new Error("LiveSet/NextPointeeId not found");
  let next = parseInt(counter.getAttribute("Value")!, 10);
  let count = 0;
  for (const tag of POINTEE_TAGS) {
    for (const el of allEl(deviceEl, tag)) {
      if (el.hasAttribute("Id")) { el.setAttribute("Id", String(next++)); count++; }
    }
  }
  counter.setAttribute("Value", String(next));
  return count;
}
```

```ts
  numAudioOuts: number;
```

```ts
import { readdirSync, statSync, existsSync } from "node:fs";
import { join, basename } from "node:path";
import { homedir } from "node:os";

export interface InventoryEntry { name: string; path: string }

export function scanVst3Dir(dir: string): InventoryEntry[] {
  if (!existsSync(dir)) return [];
  const out: InventoryEntry[] = [];
  for (const name of readdirSync(dir)) {
    const p = join(dir, name);
    if (name.toLowerCase().endsWith(".vst3")) {
      out.push({ name: basename(name, ".vst3"), path: p });      // file OR bundle dir
    } else if (statSync(p).isDirectory()) {
      out.push(...scanVst3Dir(p));                                // vendor subfolder
    }
  }
  return out;
}

export function defaultVst3Dirs(): string[] {
  if (process.platform === "win32") return ["C:\\Program Files\\Common Files\\VST3"];
  if (process.platform === "darwin")
    return ["/Library/Audio/Plug-Ins/VST3", join(homedir(), "Library/Audio/Plug-Ins/VST3")];
  return [];
}

const norm = (s: string) => s.toLowerCase().replace(/\s+/g, " ").trim();

export function inventoryHas(inv: InventoryEntry[], name: string): boolean {
  const n = norm(name);
  return inv.some(e => norm(e.name) === n);
}
```

```ts
import { auCodes } from "./devices.ts";
import type { PlistDict } from "./plist.ts";
import { lookupIdentity, resolveUid } from "./uid.ts";
import type { IdentityEntry } from "./uid.ts";
import { inventoryHas } from "./inventory.ts";
import type { InventoryEntry } from "./inventory.ts";

export type Tier = "swap" | "install" | "substitute" | "freeze" | "unknown";
export type StateAction = "transplant-juce" | "extract-fxp" | "default-state" | "none";
export interface Verdict {
  tier: Tier; stateAction: StateAction; reason: string;
  entry: IdentityEntry | null;
  uid: [number, number, number, number] | null; uidSource: string | null;
}

// Same-category substitutes for the manual tier (extensible data, spec §5).
const CATEGORY_HINTS: Record<string, string[]> = {
  clipper: ["kHs Clipper", "KClip3"],
};

export function decideAu(state: PlistDict | null, inventory: InventoryEntry[]): Verdict {
  if (!state) return { tier: "unknown", stateAction: "none", reason: "AU state unreadable — reporting only", entry: null, uid: null, uidSource: null };

  const codes = auCodes(state);
  const entry = lookupIdentity(codes.subtype, codes.manufacturer);
  const hasJuce = Buffer.isBuffer(state.jucePluginState);
  const targetName = entry?.vst3Name ?? String(state.name ?? "");
  const installed = inventoryHas(inventory, targetName) ||
    (entry ? inventoryHas(inventory, entry.vst3Name.replace(/\s+/g, "")) : false); // "Serum 2" vs "Serum2"
  const resolved = resolveUid(entry, codes.manufacturer, codes.subtype, hasJuce);

  if (installed && entry?.crossVersion && resolved)
    return { tier: "swap", stateAction: "extract-fxp", reason: `version crossing → swap to ${entry.vst3Name} + extract preset for vendor importer`, entry, uid: resolved.uid, uidSource: resolved.source };
  if (installed && resolved && hasJuce)
    return { tier: "swap", stateAction: "transplant-juce", reason: "JUCE state — byte-copy transplant", entry, uid: resolved.uid, uidSource: resolved.source };
  if (installed && resolved)
    return { tier: "swap", stateAction: "default-state", reason: "plugin installed; state format unknown — swapped with default settings", entry, uid: resolved.uid, uidSource: resolved.source };

  if (entry?.category) {
    for (const sub of CATEGORY_HINTS[entry.category] ?? []) {
      if (sub !== entry.vst3Name && inventoryHas(inventory, sub))
        return { tier: "substitute", stateAction: "none", reason: `not installed; same-category substitute owned: ${sub} (manual)`, entry, uid: null, uidSource: null };
    }
  }
  return { tier: "freeze", stateAction: "none", reason: "plugin not installed on this machine — request frozen track", entry, uid: null, uidSource: null };
}
```

```ts
import { inflateSync } from "node:zlib";

export interface SerumExtract { fxp: Buffer; presetName: string; assetPaths: string[] }

export function serumFromVstdata(vstdata: Buffer): SerumExtract {
  if (vstdata.length < 64 || vstdata.toString("latin1", 0, 4) !== "CcnK")
    throw new Error("not a Serum vstdata blob (missing CcnK magic)");
  if (vstdata.toString("latin1", 8, 12) !== "FPCh")
    throw new Error("unsupported fxp variant (expected FPCh opaque-chunk program)");

  const fxp = Buffer.from(vstdata);                   // copy; never mutate the source
  fxp.writeInt32BE(1, 24);                            // DAW-state chunks carry odd counts; presets use 1
  const presetName = fxp.toString("latin1", 28, 56).replace(/\0+$/, "");

  let assetPaths: string[] = [];
  try {
    const inner = inflateSync(fxp.subarray(60)).toString("latin1");
    assetPaths = [...new Set(inner.match(/\/[ -~]{3,200}?\.wav/g) ?? [])];
  } catch { /* not zlib — report no assets rather than fail the extract */ }

  return { fxp, presetName, assetPaths };
}
```

```ts
import { writeFileSync, mkdirSync } from "node:fs";
import { join, basename } from "node:path";
import { loadAls, saveAls } from "./als.ts";
import { scanDevices, auState, auCodes, fourcc, liveDeviceType } from "./devices.ts";
import { scanVst3Dir, defaultVst3Dirs } from "./inventory.ts";
import type { InventoryEntry } from "./inventory.ts";
import { decideAu } from "./tiers.ts";
import type { Tier, StateAction } from "./tiers.ts";
import { instantiateVst3Device } from "./skeleton.ts";
import { replaceAuDevice } from "./transplant.ts";
import { serumFromVstdata } from "./assets/serum.ts";

export interface ReportLine { track: string; device: string; tier: Tier; stateAction: StateAction; detail: string }
export interface ConvertResult { report: ReportLine[]; outPath: string | null; presets: { name: string; path: string }[] }

export function convertProject(
  alsPath: string,
  opts: { outDir: string; vst3Dirs?: string[]; write?: boolean },
): ConvertResult {
  const write = opts.write ?? true;
  const doc = loadAls(alsPath);
  const inventory: InventoryEntry[] = (opts.vst3Dirs ?? defaultVst3Dirs()).flatMap(scanVst3Dir);
  const report: ReportLine[] = [];
  const presets: { name: string; path: string }[] = [];
  let swapped = 0;

  for (const dev of scanDevices(doc).filter(d => d.kind === "au")) {
    const state = auState(dev);
    const verdict = decideAu(state, inventory);
    let detail = verdict.reason;

    if (verdict.tier === "swap" && write && state && verdict.uid) {
      const codes = auCodes(state);
      const entry = verdict.entry;
      const el = instantiateVst3Device(doc, {
        name: entry?.vst3Name ?? dev.name,
        uid: verdict.uid,
        deviceType: entry?.deviceType ?? liveDeviceType(fourcc(codes.type)),
        numAudioIns: entry?.numAudioIns ?? 2,
        numAudioOuts: entry?.numAudioOuts ?? (liveDeviceType(fourcc(codes.type)) === 1 ? 1 : 2),
      });
      const juce = verdict.stateAction === "transplant-juce" && Buffer.isBuffer(state.jucePluginState)
        ? (state.jucePluginState as Buffer) : null;
      replaceAuDevice(doc, dev.el, el, juce);
      swapped++;

      if (verdict.stateAction === "extract-fxp" && Buffer.isBuffer(state.vstdata)) {
        try {
          const ex = serumFromVstdata(state.vstdata as Buffer);
          mkdirSync(opts.outDir, { recursive: true });
          const p = join(opts.outDir, `${ex.presetName || "preset"}.fxp`);
          writeFileSync(p, ex.fxp);
          presets.push({ name: ex.presetName, path: p });
          detail += ` · preset → ${basename(p)}`;
          if (ex.assetPaths.length) detail += ` · external assets: ${ex.assetPaths.join(", ")}`;
        } catch (e) { detail += ` · preset extraction failed: ${(e as Error).message}`; }
      }
    }
    report.push({ track: dev.trackName, device: dev.name, tier: verdict.tier, stateAction: verdict.stateAction, detail });
  }

  let outPath: string | null = null;
  if (write && swapped > 0) {
    mkdirSync(opts.outDir, { recursive: true });
    outPath = join(opts.outDir, basename(alsPath).replace(/\.als$/i, "") + "-win.als");
    saveAls(doc, outPath);
  }
  return { report, outPath, presets };
}
```
